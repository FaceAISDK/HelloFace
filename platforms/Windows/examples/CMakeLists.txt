cmake_minimum_required(VERSION 3.22.1)
project(HiFaceSDKExample LANGUAGES C CXX)
find_package(HiFace CONFIG REQUIRED COMPONENTS C)
add_executable(hiface_infrared_liveness infrared_liveness.c)
target_link_libraries(hiface_infrared_liveness PRIVATE HiFace::HiFace)
add_executable(hiface_c_consumer c_consumer.c)
target_link_libraries(hiface_c_consumer PRIVATE HiFace::HiFace)
add_executable(hiface_cpp_consumer cpp_consumer.cpp)
target_link_libraries(hiface_cpp_consumer PRIVATE HiFace::HiFace)
add_executable(hiface_model_consumer model_consumer.cpp)
target_link_libraries(hiface_model_consumer PRIVATE HiFace::HiFace)
foreach(app hiface_c_consumer hiface_cpp_consumer hiface_model_consumer hiface_infrared_liveness)
    set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
endforeach()
file(GLOB runtime_dlls "${HiFace_RUNTIME_DIR}/*.dll")
add_custom_target(hiface_example_assets
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_BINARY_DIR}/bin/$<CONFIG>"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${runtime_dlls} "${CMAKE_BINARY_DIR}/bin/$<CONFIG>" VERBATIM)
add_dependencies(hiface_c_consumer hiface_example_assets)
add_dependencies(hiface_cpp_consumer hiface_example_assets)
add_dependencies(hiface_model_consumer hiface_example_assets)
add_dependencies(hiface_infrared_liveness hiface_example_assets)
